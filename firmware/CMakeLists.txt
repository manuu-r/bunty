cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32s3)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/src" "${CMAKE_CURRENT_LIST_DIR}/lib")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/scripts/prepare_deps.py"
                RESULT_VARIABLE BUNTY_DEPS_RESULT)
if(NOT BUNTY_DEPS_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not prepare Bunty dependencies")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(bunty VERSION 0.1.0)
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
set(BUNTY_IDF_VERSION "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}")
if(NOT BUNTY_IDF_VERSION STREQUAL "6.0.1")
    message(FATAL_ERROR "Bunty requires ESP-IDF 6.0.1; found ${BUNTY_IDF_VERSION}")
endif()
